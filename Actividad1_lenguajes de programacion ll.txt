-- Crear BD
CREATE DATABASE BD_UNI;
GO
USE BD_UNI;
GO

-- 1. Tabla CENTRO (Catálogo)
CREATE TABLE CENTRO (
    NumeroCentro CHAR(6) PRIMARY KEY,
    NombreCentro VARCHAR(100) NOT NULL,
    Ciudad VARCHAR(50) NOT NULL
);

-- 2. Tabla EMPLEADO (Tabla Padre)
CREATE TABLE EMPLEADO (
    NumeroEmpleado INT IDENTITY(1,1) PRIMARY KEY,
    Nombre VARCHAR(50) NOT NULL,
    ApellidoPaterno VARCHAR(50) NOT NULL,
    ApellidoMaterno VARCHAR(50) NOT NULL,
    FechaNacimiento DATE NOT NULL,
    RFC VARCHAR(13) NOT NULL, -- Se calcula: 2 ApPat + 1 ApMat + 1 Nombre + AAMMDD + XXX
    Puesto VARCHAR(50) NOT NULL,
    DescripcionPuesto VARCHAR(100) NOT NULL,
    CentroTrabajo CHAR(6) NOT NULL,
    EsDirectivo BIT NOT NULL DEFAULT 0,
    CONSTRAINT FK_Empleado_Centro FOREIGN KEY (CentroTrabajo) REFERENCES CENTRO(NumeroCentro)
);

-- 3. Tabla DIRECTIVO (Tabla Hija - Herencia)
CREATE TABLE DIRECTIVO (
    NumeroEmpleado INT PRIMARY KEY,
    NumeroCentroSupervisa CHAR(6) NOT NULL,
    PrestacionCombustible BIT NOT NULL DEFAULT 0,
    CONSTRAINT FK_Directivo_Empleado FOREIGN KEY (NumeroEmpleado) REFERENCES EMPLEADO(NumeroEmpleado),
    CONSTRAINT FK_Directivo_CentroSupervisa FOREIGN KEY (NumeroCentroSupervisa) REFERENCES CENTRO(NumeroCentro)
);
